# The 16 bit compute launch grid, on the dense scene from seq_16bit_dispatch.cfg.
#
# USE_16BIT_DISPATCH changes the shader source of every compute pass that is launched from
# a linear work item count - traversal, build setup, streaming, and the compute rasterizer -
# and normally never compiles at all on hardware whose maxComputeWorkGroupCount is large.
#
# Not a full matrix on purpose: what has to be covered is the set of passes that carry the
# macro, not the feature permutations around them. Those are already covered by
# seq_visualize.txt, seq_raster_features.txt and seq_raytrace_blas.txt.

SEQUENCE "d16_raster_hw"
--sequenceframes 32
--sequenceresetframes 8
--renderer 0
--swraster 0
--culling 1
--twopassculling 1
--discretelod 0
--visualize 0

# the compute rasterizer is the pass with the most 16 bit specific shader code
SEQUENCE "d16_raster_sw"
--renderer 0
--swraster 1
--culling 1
--twopassculling 1
--discretelod 0
--visualize 2

SEQUENCE "d16_raster_sw_twopass_off"
--renderer 0
--swraster 1
--culling 1
--twopassculling 0
--discretelod 0
--visualize 2

SEQUENCE "d16_raster_discretelod"
--renderer 0
--swraster 0
--culling 1

--visualize 11
