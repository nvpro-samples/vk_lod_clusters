# Hybrid path on the material fixture: alpha mask and blend, two sided, textures, and the
# raster state the hybrid raster pass shares with the raster renderer.
#
# Every configuration runs twice: from a static close view into the grid, where most
# instances are hidden behind the nearest bunnies, and as `_motion` along a sideways
# camera path, which is what reaches the second raster pass (disocclusion). The static
# view is camera path 1, a held path overrides --camerastring.
# --hybridmatchcheck traces the primary ray as well and logs an error when more pixels
# disagree with the raster than edge rounding explains.

SEQUENCE "hy_mat_default"
--sequenceframes 32
--sequenceresetframes 8
--renderer 3
--hybridmatchcheck 1
--dlss 0
--culling 1
--streaming 1
--gridunique 0
--blassharing 1
--blasmerging 1
--blascaching 1
--hitstreaming 0
--addcamerapath "smooth 1 loop 1 dur 4 ; {-20, 6, 20}, {0, 0, 0}, {0, 1, 0}, {60} ; {20, 6, 20}, {0, 0, 0}, {0, 1, 0}, {60} ; {-20, 6, 20}, {0, 0, 0}, {0, 1, 0}, {60}"
--addcamerapath "smooth 0 loop 0 dur 1 ; {-14, 5, 14}, {0, 0, 0}, {0, 1, 0}, {60} ; {-14, 5, 14}, {0, 0, 0}, {0, 1, 0}, {60}"
--forcetwosided 0
--flipwinding 0
--extmeshshader 0
--runcamerapath 1 1

SEQUENCE "hy_mat_default_motion"
--runcamerapath 0 32

SEQUENCE "hy_mat_forcetwosided"
--forcetwosided 1
--flipwinding 0
--extmeshshader 0
